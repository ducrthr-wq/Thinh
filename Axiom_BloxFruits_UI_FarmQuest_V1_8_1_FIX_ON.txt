--=========================================================
-- EMPTY TAB SYSTEM
-- Chỉ giữ giao diện, không có chức năng
--=========================================================

local tabs = {}
local tabButtons = {}

local function createTab(name)
    local page = Instance.new("Frame")
    page.Name = name .. "Page"
    page.Size = UDim2.new(1, 0, 1, 0)
    page.BackgroundTransparency = 1
    page.Visible = false
    page.Parent = Content

    local emptyText = Instance.new("TextLabel")
    emptyText.Size = UDim2.new(1, 0, 1, 0)
    emptyText.BackgroundTransparency = 1
    emptyText.Text = ""
    emptyText.Parent = page

    local button = Instance.new("TextButton")
    button.Size = UDim2.new(1, -16, 0, 38)
    button.BackgroundColor3 = Color3.fromRGB(30, 30, 36)
    button.Text = name
    button.TextColor3 = Color3.fromRGB(205, 205, 215)
    button.TextSize = 14
    button.Font = Enum.Font.GothamMedium
    button.BorderSizePixel = 0
    button.Parent = Sidebar

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 8)
    corner.Parent = button

    tabs[name] = page
    tabButtons[name] = button

    return page
end

local function switchTab(tabName)
    for name, page in pairs(tabs) do
        page.Visible = (name == tabName)

        if tabButtons[name] then
            TweenService:Create(
                tabButtons[name],
                TweenInfo.new(0.15),
                {
                    BackgroundColor3 = name == tabName
                        and Color3.fromRGB(52, 52, 63)
                        or Color3.fromRGB(30, 30, 36)
                }
            ):Play()
        end
    end
end

local FarmPage = createTab("Farm")
local QuestPage = createTab("Quest")
local CombatPage = createTab("Combat")
local TeleportPage = createTab("Teleport")
local SettingsPage = createTab("Settings")

for name, button in pairs(tabButtons) do
    button.MouseButton1Click:Connect(function()
        switchTab(name)
    end)
end

switchTab("Farm")

local TravelState = {
    Tween = nil,
    Active = false,
    TargetKey = nil,
    TargetPosition = nil,
    Speed = 325,
}

local function cancelTravel()
    if TravelState.Tween then
        pcall(function()
            TravelState.Tween:Cancel()
        end)
    end

    TravelState.Tween = nil
    TravelState.Active = false
    TravelState.TargetKey = nil
    TravelState.TargetPosition = nil
end

local function trySpecialEntrance(q)
    if not q then
        return
    end

    local remote = getCommRemote and getCommRemote() or nil
    if not remote then
        local remotes = ReplicatedStorage:FindFirstChild("Remotes")
        remote = remotes and (
            remotes:FindFirstChild("CommF_")
            or remotes:FindFirstChild("CommF")
        )
    end

    if not remote then
        return
    end

    local mob = q.Mob

    pcall(function()
        if mob == "Fishman Warrior"
            or mob == "Fishman Commando" then

            remote:InvokeServer(
                "requestEntrance",
                Vector3.new(61163.8516, 11.6797, 1819.7842)
            )

        elseif mob == "God's Guard" then
            remote:InvokeServer(
                "requestEntrance",
                Vector3.new(-4607.8228, 872.5425, -1667.5569)
            )

        elseif mob == "Shanda"
            or mob == "Royal Squad"
            or mob == "Royal Soldier" then

            remote:InvokeServer(
                "requestEntrance",
                Vector3.new(-7894.6177, 5547.1416, -380.2912)
            )

        elseif mob == "Ship Deckhand"
            or mob == "Ship Engineer"
            or mob == "Ship Steward"
            or mob == "Ship Officer" then

            remote:InvokeServer(
                "requestEntrance",
                Vector3.new(923.2125, 126.9760, 32852.8320)

end

local function findReplicatedMobPosition(wantedName)
    if not wantedName then
        return nil
    end

    local wanted = string.lower(wantedName)

    for _, obj in ipairs(ReplicatedStorage:GetChildren()) do
        if obj:IsA("Model")
            and string.find(
                string.lower(obj.Name),
                wanted,
                1,
                true
            ) then

            local root =
                obj:FindFirstChild("HumanoidRootPart")
                or obj:FindFirstChild("Torso")
                or obj.PrimaryPart

            if root and root:IsA("BasePart") then
                return root.CFrame
            end
        end
    end

    return nil
end

local function getMobAreaCFrame(q)
    if not q then
        return nil

    return MOB_SPAWNS[q.Mob]
        or findReplicatedMobPosition(q.Mob)
end

local function travelToMobArea(root, humanoid, q)
    if not root or not root.Parent or not q then
        return false
    end

    local area = getMobAreaCFrame(q)
    if not area then
        return false
    end

    trySpecialEntrance(q)

    -- Đứng cao hơn bãi mob để lúc mob stream vào không bị đánh ngay.
    local target =
        area
        * CFrame.new(0, math.max(FarmState.HoverHeight, 25), 0)

    local distance = (root.Position - target.Position).Magnitude

    if distance <= 90 then
        cancelTravel()
        return true
    end

    if TravelState.Active
        and TravelState.TargetKey == q.Mob
        and TravelState.TargetPosition
        and (TravelState.TargetPosition - target.Position).Magnitude <= 5 then

        return false
    end

    cancelTravel()

    TravelState.Active = true
    TravelState.TargetKey = q.Mob
    TravelState.TargetPosition = target.Position

    pcall(function()
        if humanoid then
            humanoid.Sit = false
            humanoid.AutoRotate = false
        end
    end)

    local duration = math.clamp(
        distance / TravelState.Speed,
        0.15,
        18
    )

    local tween = TweenService:Create(
        root,
        TweenInfo.new(
            duration,
            Enum.EasingStyle.Linear,
            Enum.EasingDirection.Out
        ),
        {CFrame = target}
    )

    TravelState.Tween = tween

    tween.Completed:Connect(function()
        if TravelState.Tween == tween then
            TravelState.Tween = nil
            TravelState.Active = false
        end

        pcall(function()
            if humanoid then
                humanoid.AutoRotate = true
            end
        end)
    end)

    tween:Play()
    return false
end

    return nil
end

local function getQuestEnemies(fromPosition, wantedName, maxTargets)
    local enemiesFolder = workspace:FindFirstChild("Enemies")
    if not enemiesFolder then
        return {}
    end

    local wanted = wantedName and string.lower(wantedName) or nil
    local found = {}

    for _, enemy in ipairs(enemiesFolder:GetChildren()) do
        if enemy:IsA("Model") then
            local humanoid = enemy:FindFirstChildOfClass("Humanoid")
            local root =
                enemy:FindFirstChild("HumanoidRootPart")
                or enemy:FindFirstChild("Torso")

            if humanoid and root and humanoid.Health > 0 then
                local nameOK = true

                if wanted then
                    nameOK = string.find(
                        string.lower(enemy.Name),
                        wanted,
                        1,
                        true
                    ) ~= nil
                end

                if nameOK then
                    table.insert(found, {
                        Enemy = enemy,
                        Humanoid = humanoid,
                        Root = root,
                        Distance = (root.Position - fromPosition).Magnitude,
                    })
                end
            end
        end
    end

    table.sort(found, function(a, b)
        return a.Distance < b.Distance
    end)

    local limit = math.min(maxTargets or #found, #found)
    local result = {}

    for i = 1, limit do
        result[i] = found[i]
    end

    return result
end

local function tryExpandSimulationRadius()
    pcall(function()
        if type(sethiddenproperty) == "function" then
            sethiddenproperty(Player, "SimulationRadius", math.huge)
        elseif type(setsimulationradius) == "function" then
            setsimulationradius(math.huge, math.huge)
        end
    end)
end

local function magnetQuestEnemies(targets, anchorCFrame)
    if not FarmState.MagnetEnabled
        or not anchorCFrame
        or type(targets) ~= "table" then

        return
    end

    tryExpandSimulationRadius()

    for _, data in ipairs(targets) do
        local enemy = data.Enemy
        local humanoid = data.Humanoid
        local root = data.Root

        if enemy and enemy.Parent
            and humanoid and humanoid.Health > 0
            and root and root.Parent then

            local distanceToAnchor =
                (root.Position - anchorCFrame.Position).Magnitude

            if distanceToAnchor <= FarmState.MagnetRadius then
                pcall(function()
                    root.CanCollide = false
                    root.AssemblyLinearVelocity = Vector3.zero
                    root.AssemblyAngularVelocity = Vector3.zero

                    -- Gom quái đúng quest về cùng 1 điểm.
                    root.CFrame = anchorCFrame

                    -- Giảm khả năng mob chạy tản ra.
                    humanoid.WalkSpeed = 0
                    humanoid.JumpPower = 0
                    humanoid:ChangeState(Enum.HumanoidStateType.Physics)
                end)
            end
        end
    end
end

local function toolPriority(tool)
    if not tool:IsA("Tool") then
        return -1
    end

    local tip = string.lower(tostring(tool.ToolTip or ""))

    if tip == "melee" or tip:find("melee", 1, true) then
        return 300
    end

    if tip == "sword" or tip:find("sword", 1, true) then
        return 200
    end

    return 100
end

local function findBestTool(character)
    local equipped = character:FindFirstChildOfClass("Tool")
    if equipped then
        return equipped
    end

    local backpack = Player:FindFirstChildOfClass("Backpack")
        or Player:FindFirstChild("Backpack")

    if not backpack then
        return nil
    end

    local bestTool = nil
    local bestPriority = -1

    for _, child in ipairs(backpack:GetChildren()) do
        local priority = toolPriority(child)

        if priority > bestPriority then
            bestPriority = priority
            bestTool = child
        end
    end

    return bestTool
end

local function equipBestTool(character, humanoid)
    local tool = findBestTool(character)

    if tool and tool.Parent ~= character then
        pcall(function()
            humanoid:EquipTool(tool)
        end)
    end

    if tool and tool.Parent == character then
        return tool
    end

    return character:FindFirstChildOfClass("Tool")
end

--
-- Không auto-click.
-- Không cần đánh từng mob riêng lẻ.
--
-- Luồng:
--   RegisterAttack:FireServer(delay)
--   RegisterHit:FireServer(BasePart, OthersEnemies)

--   ...

local MultiAttack = {
    Ready = false,
    Error = nil,
    RegisterAttack = nil,
    RegisterHit = nil,
    LastAttack = 0,
    Delay = 0.14,
}

local function initMultiAttack()
    if MultiAttack.Ready then
        return true
    end

    local modules = ReplicatedStorage:FindFirstChild("Modules")
    local net = modules and modules:FindFirstChild("Net")

    if not net then
        MultiAttack.Error = "NO_MODULES_NET"
        return false
    end

    local registerAttack = net:FindFirstChild("RE/RegisterAttack")
    local registerHit = net:FindFirstChild("RE/RegisterHit")

    if not registerAttack then
        MultiAttack.Error = "NO_RE_REGISTER_ATTACK"
        return false
    end

    if not registerHit then
        MultiAttack.Error = "NO_RE_REGISTER_HIT"
        return false
    end

    MultiAttack.RegisterAttack = registerAttack
    MultiAttack.RegisterHit = registerHit
    MultiAttack.Ready = true
    MultiAttack.Error = nil

    return true
end

local function getPreferredHitPart(enemy)
    if not enemy or not enemy.Parent then
        return nil
    end

    local preferred = {
        "Head",
        "HumanoidRootPart",
        "UpperTorso",
        "Torso",
        "RightLowerLeg",
        "LeftLowerLeg",
    }

    for _, name in ipairs(preferred) do
        local part = enemy:FindFirstChild(name)
        if part and part:IsA("BasePart") then
            return part
        end
    end

    return enemy:FindFirstChildWhichIsA("BasePart", true)
end

local function multiMobAttack(tool, targetData, playerRoot)
    if not tool or not tool.Parent then
        return false, "NO_TOOL", 0
    end

    if not playerRoot then
        return false, "NO_PLAYER_ROOT", 0
    end

    if not initMultiAttack() then
        return false, MultiAttack.Error or "NET_INIT_FAILED", 0
    end

    local now = os.clock()

    if now - MultiAttack.LastAttack < MultiAttack.Delay then
        return false, "ATTACK_COOLDOWN", 0
    end

    local others = {}
    local basePart = nil

    for _, data in ipairs(targetData) do
        local enemy = data.Enemy
        local humanoid = data.Humanoid

        if enemy and enemy.Parent
            and humanoid and humanoid.Health > 0 then

            local part = getPreferredHitPart(enemy)

            if part then
                local distance =
                    (part.Position - playerRoot.Position).Magnitude

                if distance <= FarmState.HitDistance then
                    basePart = basePart or part
                    table.insert(others, {enemy, part})
                end
            end
        end
    end

    if not basePart or #others == 0 then
        return false, "NO_TARGETS_IN_RANGE", 0
    end

    MultiAttack.LastAttack = now

    local attackOK = pcall(function()
        MultiAttack.RegisterAttack:FireServer(0)
    end)

    if not attackOK then
        return false, "REGISTER_ATTACK_FAILED", 0
    end

    local hitOK = pcall(function()
        MultiAttack.RegisterHit:FireServer(
            basePart,
            others
        )
    end)

    if not hitOK then
        return false, "REGISTER_HIT_FAILED", 0
    end

    return true, "MULTI_NET_HIT", #others
end

local function safeAttack(tool, targets, playerRoot)
    return multiMobAttack(tool, targets, playerRoot)
end

local function moveToQuest(root, q)
    if not root or not q then
        return false
    end

    pcall(function()
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        root.CFrame = q.CFrame * CFrame.new(0, 2.5, 0)
    end)

    return (root.Position - q.CFrame.Position).Magnitude <= 20
end

local function requestQuest(q, root)
    if not q or not root then
        return false
    end

    if os.clock() - FarmState.LastQuestAttempt < FarmState.QuestRetryDelay then
        return false
    end

    FarmState.LastQuestAttempt = os.clock()

    local level = getPlayerLevel()
    tryTravelToSubmerged(level)

    moveToQuest(root, q)
    task.wait(0.22)

    local remote = getCommRemote()
    if not remote then
        return false
    end

    local ok = pcall(function()
        remote:InvokeServer("StartQuest", q.Quest, q.Index)
    end)

    task.wait(0.28)

    return ok
end

local function abandonQuest()
    local remote = getCommRemote()
    if not remote then
        return
    end

    pcall(function()
        remote:InvokeServer("AbandonQuest")
    end)
end

local function startFarm()
    if FarmState.Enabled then
        return
    end

    FarmState.Enabled = true
    FarmState.Token += 1
    FarmState.LastQuestAttempt = 0

    local myToken = FarmState.Token
    local lastAttack = 0

    updateToggleVisual()
    FarmStatus.Text = "Trạng thái: Khởi động Farm..."

    task.spawn(function()
        local function farmLoop()
            while FarmState.Enabled
                and FarmState.Token == myToken
                and ScreenGui.Parent do

                local character, humanoid, playerRoot = getCharacter()

                if not character then
                    FarmStatus.Text = "Trạng thái: Đợi nhân vật..."
                    task.wait(0.25)
                    continue
                end

                local level = getPlayerLevel()
                local q = resolveQuest(level)

                if not q then
                    FarmStatus.Text =
                        "Không có dữ liệu quest cho Lv." .. tostring(level)

                    task.wait(0.5)
                    continue
                end

                --=================================================
                -- 1) AUTO QUEST
                --=================================================

                if not isQuestVisible() then
                    cancelTravel()

                    FarmStatus.Text =
                        "Nhận quest Lv." .. tostring(level) ..
                        ": " .. q.Mob

                    requestQuest(q, playerRoot)

                    task.wait(0.18)
                    continue
                end

                if not questMatches(q) then
                    cancelTravel()

                    FarmStatus.Text =
                        "Quest sai → nhận lại: " .. q.Mob

                    abandonQuest()

                    task.wait(0.35)
                    continue
                end

                --=================================================
                -- 2) TÌM MOB ĐÚNG QUEST
                --=================================================

                local targets = getQuestEnemies(
                    playerRoot.Position,
                    q.Mob,
                    FarmState.MaxTargets
                )

                --=================================================
                -- 3) MOB CHƯA LOAD -> BAY TỚI BÃI MOB
                --=================================================

                if #targets == 0 then
                    local mobArea = getMobAreaCFrame(q)

                    if mobArea then
                        local distance =
                            (playerRoot.Position - mobArea.Position).Magnitude

                        if distance > 110 then
                            FarmStatus.Text =
                                "Bay tới " .. q.Mob ..
                                " | " .. tostring(math.floor(distance)) ..
                                " studs"

                            travelToMobArea(
                                playerRoot,
                                humanoid,
                                q
                            )
                        else
                            cancelTravel()

                            FarmStatus.Text =
                                "Đã tới bãi " .. q.Mob ..
                                " | Đợi mob load/spawn..."
                        end
                    else
                        cancelTravel()

                        FarmStatus.Text =
                            "Thiếu vị trí bãi: " .. q.Mob
                    end

                    task.wait(0.12)
                    continue
                end

                --=================================================
                -- 4) MOB ĐÃ LOAD -> DỪNG BAY, GOM QUÁI
                --=================================================

                cancelTravel()

                local primary = targets[1]

                if not primary then
                    task.wait(0.08)
                    continue
                end

                local enemy = primary.Enemy
                local enemyHumanoid = primary.Humanoid
                local enemyRoot = primary.Root

                if not enemy
                    or not enemy.Parent
                    or not enemyHumanoid
                    or enemyHumanoid.Health <= 0
                    or not enemyRoot
                    or not enemyRoot.Parent then

                    task.wait(0.08)
                    continue
                end

                local magnetCFrame = enemyRoot.CFrame

                magnetQuestEnemies(
                    targets,
                    magnetCFrame
                )

                --=================================================
                -- 5) ĐỨNG TRÊN CỤM QUÁI
                --=================================================

                local farmPosition =
                    magnetCFrame.Position
                    + Vector3.new(
                        0,
                        FarmState.HoverHeight,
                        0
                    )

                local targetCFrame = CFrame.lookAt(
                    farmPosition,
                    magnetCFrame.Position
                )

                pcall(function()
                    playerRoot.AssemblyLinearVelocity =
                        Vector3.zero

                    playerRoot.AssemblyAngularVelocity =
                        Vector3.zero

                    playerRoot.CFrame = targetCFrame
                end)

                --=================================================
                -- 6) EQUIP + MULTI HIT
                --=================================================

                local tool =
                    equipBestTool(character, humanoid)

                local now = os.clock()

                if tool
                    and (now - lastAttack)
                        >= FarmState.AttackDelay then

                    lastAttack = now

                    local ok, backend, hitCount =
                        safeAttack(
                            tool,
                            targets,
                            playerRoot
                        )

                    if ok then
                        FarmStatus.Text =
                            "Lv." .. tostring(level) ..
                            " | " .. q.Mob ..
                            " | x" .. tostring(hitCount or 0) ..
                            " | " .. tostring(backend)
                    elseif backend ~= "ATTACK_COOLDOWN" then
                        FarmStatus.Text =
                            "Combat: " .. tostring(backend) ..
                            " | Mob: " .. tostring(#targets)
                    end

                elseif not tool then
                    FarmStatus.Text =
                        "Không tìm thấy Tool để đánh"
                end

                task.wait(FarmState.UpdateDelay)
            end
        end

        -- Nếu có lỗi runtime, không để toggle chết im.
        local ok, err = xpcall(
            farmLoop,
            function(message)
                return tostring(message)
            end
        )

        if not ok then
            FarmStatus.Text =
                "Farm lỗi: " .. tostring(err)

            warn(
                "[AXIOM FARM ERROR]",
                tostring(err)
            )

            FarmState.Enabled = false
            FarmState.Token += 1
            cancelTravel()
            updateToggleVisual()
        end
    end)
end

--=========================================================
-- DRAG SYSTEM
--=========================================================

local dragging = false
local dragStart
local startPos

TopBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)

TopBar.InputChanged:Connect(function(input)
    if dragging and (
        input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch
    ) then

        local delta = input.Position - dragStart

        Main.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = false
    end
end)
