local function createTab(name)
    local page = Instance.new("Frame")
    page.Name = name .. "Page"
    page.Size = UDim2.new(1, 0, 1, 0)
    page.BackgroundTransparency = 1
    page.Visible = false
    page.Parent = Content

    local emptyText = Instance.new("TextLabel")
    emptyText.Size = UDim2.new(1, 0, 1, 0)
    emptyText.BackgroundTransparency = 1
    emptyText.Text = ""
    emptyText.Parent = page

    local button = Instance.new("TextButton")
    button.Size = UDim2.new(1, -16, 0, 38)
    button.BackgroundColor3 = Color3.fromRGB(30, 30, 36)
    button.Text = name
    button.TextColor3 = Color3.fromRGB(205, 205, 215)
    button.TextSize = 14
    button.Font = Enum.Font.GothamMedium
    button.BorderSizePixel = 0
    button.Parent = Sidebar

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 8)
    corner.Parent = button

    tabs[name] = page
    tabButtons[name] = button

    return page
end

local function switchTab(tabName)
    for name, page in pairs(tabs) do
        page.Visible = (name == tabName)

        if tabButtons[name] then
            TweenService:Create(
                tabButtons[name],
                TweenInfo.new(0.15),
                {
                    BackgroundColor3 = name == tabName
                        and Color3.fromRGB(52, 52, 63)
                        or Color3.fromRGB(30, 30, 36)
                }
            ):Play()
        end
    end
end

local FarmState = {
    Enabled = false,
    Token = 0,
    Height = 2.6,
    BackOffset = 2.8,
    UpdateDelay = 0.06,
    AttackDelay = 0.12,
}

local function getCharacter()
    local character = Player.Character
    if not character or not character.Parent then
        return nil
    end

    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local root = character:FindFirstChild("HumanoidRootPart")

    if not humanoid or humanoid.Health <= 0 or not root then
        return nil
    end

    return character, humanoid, root
end

local function getNearestEnemy(fromPosition)
    local enemies = workspace:FindFirstChild("Enemies")
    if not enemies then
        return nil
    end

    local nearest = nil
    local nearestDistance = math.huge

    for _, enemy in ipairs(enemies:GetChildren()) do
        if enemy:IsA("Model") then
            local humanoid = enemy:FindFirstChildOfClass("Humanoid")
            local root = enemy:FindFirstChild("HumanoidRootPart")
                or enemy:FindFirstChild("Torso")

            if humanoid and root and humanoid.Health > 0 then
                local distance = (root.Position - fromPosition).Magnitude

                if distance < nearestDistance then
                    nearest = enemy
                    nearestDistance = distance
                end
            end
        end
    end

    return nearest, nearestDistance
end

local function toolPriority(tool)
    if not tool:IsA("Tool") then
        return -1
    end

    local tip = string.lower(tostring(tool.ToolTip or ""))

    if tip == "melee" or tip:find("melee", 1, true) then
        return 300
    end

    if tip == "sword" or tip:find("sword", 1, true) then
        return 200
    end

    return 100
end

local function findBestTool(character)
    local equipped = character:FindFirstChildOfClass("Tool")
    if equipped then
        return equipped
    end

    local backpack = Player:FindFirstChildOfClass("Backpack")
        or Player:FindFirstChild("Backpack")

    if not backpack then
        return nil
    end

    local bestTool = nil
    local bestPriority = -1

    for _, child in ipairs(backpack:GetChildren()) do
        local priority = toolPriority(child)

        if priority > bestPriority then
            bestPriority = priority
            bestTool = child
        end
    end

    return bestTool
end

local function equipBestTool(character, humanoid)
    local tool = findBestTool(character)

    if tool and tool.Parent ~= character then
        pcall(function()
            humanoid:EquipTool(tool)
        end)
    end

    if tool and tool.Parent == character then
        return tool
    end

    return character:FindFirstChildOfClass("Tool")
end

local VirtualInputManager = game:GetService("VirtualInputManager")

local function safeAttack(tool)
    if not tool or not tool.Parent then
        return false
    end

    local attacked = false

    -- Ưu tiên mô phỏng click thật để đi qua luồng input của game.
    pcall(function()
        local camera = workspace.CurrentCamera
        local viewport = camera and camera.ViewportSize or Vector2.new(800, 600)
        local x = math.floor(viewport.X * 0.5)
        local y = math.floor(viewport.Y * 0.5)

        VirtualInputManager:SendMouseButtonEvent(
            x, y,
            0,
            true,
            game,
            0
        )

        task.wait(0.025)

        VirtualInputManager:SendMouseButtonEvent(
            x, y,
            0,
            false,
            game,
            0
        )

        attacked = true
    end)

    -- Fallback cho executor/game path nào vẫn nhận Tool:Activate().
    pcall(function()
        if tool.Enabled ~= false then
            tool:Activate()
            attacked = true
        end
    end)

    return attacked
end

local function updateToggleVisual()
    Toggle.Text = FarmState.Enabled and "ON" or "OFF"

    TweenService:Create(
        Toggle,
        TweenInfo.new(0.15),
        {
            BackgroundColor3 = FarmState.Enabled
                and Color3.fromRGB(63, 122, 82)
                or Color3.fromRGB(65, 65, 75)
        }
    ):Play()

    if not FarmState.Enabled then
        FarmStatus.Text = "Trạng thái: Tắt"
    end
end

local function stopFarm()
    FarmState.Enabled = false
    FarmState.Token += 1
    updateToggleVisual()
end

local function startFarm()
    if FarmState.Enabled then
        return
    end

    FarmState.Enabled = true
    FarmState.Token += 1

    local myToken = FarmState.Token
    local lastAttack = 0

    updateToggleVisual()

    task.spawn(function()
        while FarmState.Enabled
            and FarmState.Token == myToken
            and ScreenGui.Parent do

            local character, humanoid, playerRoot = getCharacter()

            if not character then
                FarmStatus.Text = "Trạng thái: Đợi nhân vật..."
                task.wait(0.25)
                continue
            end

            local enemy = getNearestEnemy(playerRoot.Position)

            if not enemy then
                FarmStatus.Text = "Trạng thái: Không thấy quái"
                task.wait(0.25)
                continue
            end

            local enemyHumanoid = enemy:FindFirstChildOfClass("Humanoid")
            local enemyRoot = enemy:FindFirstChild("HumanoidRootPart")
                or enemy:FindFirstChild("Torso")

            if not enemyHumanoid or not enemyRoot or enemyHumanoid.Health <= 0 then
                task.wait(FarmState.UpdateDelay)
                continue
            end

            FarmStatus.Text = "Mục tiêu: " .. enemy.Name

            -- Đứng sát phía sau/trên nhẹ để melee nằm trong tầm đánh.
            -- CFrame theo hướng của mob giúp vị trí ổn định hơn CFrame world.
            local targetCFrame =
                enemyRoot.CFrame
                * CFrame.new(0, FarmState.Height, FarmState.BackOffset)

            -- Quay mặt về mob.
            local farmPosition = targetCFrame.Position
            targetCFrame = CFrame.lookAt(
                farmPosition,
                enemyRoot.Position
            )

            pcall(function()
                playerRoot.AssemblyLinearVelocity = Vector3.zero
                playerRoot.AssemblyAngularVelocity = Vector3.zero
                playerRoot.CFrame = targetCFrame
            end)

            local tool = equipBestTool(character, humanoid)

            local now = os.clock()
            if tool and (now - lastAttack) >= FarmState.AttackDelay then
                lastAttack = now

                local ok = safeAttack(tool)

                if ok then
                    FarmStatus.Text = "Đang đánh: " .. enemy.Name
                else
                    FarmStatus.Text = "Không gửi được input đánh"
                end
            elseif not tool then
                FarmStatus.Text = "Không tìm thấy Tool để đánh"
            end

            task.wait(FarmState.UpdateDelay)
        end
    end)
end

Toggle.MouseButton1Click:Connect(function()
    if FarmState.Enabled then
        stopFarm()
    else
        startFarm()
    end
end)

--=========================================================
-- DRAG SYSTEM
--=========================================================

local dragging = false
local dragStart
local startPos

TopBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)

TopBar.InputChanged:Connect(function(input)
    if dragging and (
        input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch
    ) then

        local delta = input.Position - dragStart

        Main.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = false
    end
end)
